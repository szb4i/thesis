\begin{figure}[htpb]
	\begin{tabular}{c}
	\begin{lstlisting}[language=Python]
import numpy as np		
img = read_image(...)
u = np.fft.fftfreq(img.shape[1])
v = np.fft.fftfreq(img.shape[0])
vv, uu = np.meshgrid(v, u, indexing='ij')
D_0 = 0.05
filter_f = (np.hypot(vv, uu) < D_0)
img_f = np.fft.fft2(img)
img_filtered = np.fft.ifft2(img_ft*filter_f).real
	\end{lstlisting}
	\end{tabular}
	\caption[Example listing]{Implementation and application of \gls{ilpf} (Equation \ref{eq:ilpf}). First, a coordinate system is generated with the discrete Fourier transform sample frequencies $u$ and $v$ matching the shape of the input image. Then, the function $np.meshgrid$ creates coordinate arrays for the $v$ and the $u$ coordinates and saves them into variable $vv$ and $uu$. Finally, \gls{ilpf} is calculated in frequency domain and stored in variable \emph{filter\textunderscore f}.}\label{code:ilpf}
\end{figure}

import numpy as np		
img = read_image(...)
v = np.fft.fftfreq(img.shape[0])
u = np.fft.fftfreq(img.shape[1])
vv, uu = np.meshgrid(v, u, indexing='ij')
sigma = 1/10
filter_f = np.exp(-np.hypot(uu, vv)**2/(2*sigma**2))
img_f = np.fft.fft2(img)
img_filtered = np.fft.ifft2(img_f*filter_f).real
	\end{lstlisting}
