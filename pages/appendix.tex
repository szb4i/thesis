import numpy as np
def get_gaussian_kernel(l=3, sig=1., normalize=True):
    ax = np.linspace(-(l - 1) / 2., (l - 1) / 2., l)
    gauss = np.exp(-0.5 * np.square(ax) / np.square(sig))
    kernel = np.outer(gauss, gauss)
    if normalize:
        return kernel / np.sum(kernel)
    return kernel
	\end{lstlisting}
	\end{tabular}
	\caption[Example listing]{Definition of \emph{get\textunderscore gaussian\textunderscore kernel}}

\begin{figure}[htpb]

	\begin{lstlisting}[language=Python]
import numpy as np
img = read_image(...)
C = 9*np.identity(9)+np.full((9,9), -1)
normalizing_const = 1/(9**2)
kernel_dx = 1
kernel_dy = 1
img_wl = np.zeros((img.shape[0], img.shape[1]))
for x in range(kernel_dy, img.shape[1] - kernel_dy):
	for y in range(kernel_dx, img.shape[0] - kernel_dx):
		region_flatten = img[
			x - kernel_dx: x + kernel_dx + 1,
			y - kernel_dy: y + kernel_dy + 1
		].flatten()
		img_wl[x,y] = normalizing_const*(
			region_flatten.dot(C.dot(region_flatten))
		)
	\end{lstlisting}
	\end{tabular}

	\caption[Example listing]{Visualization of the Fourier transform. Variable \emph{img\textunderscore ft} is the discrete Fourier transform of the image. Furthermore, logarithm function needs to be applied in order to see any information on the output. Otherwise, the output is completely dark. This stage is variable \emph{img\textunderscore ft\textunderscore abs}. The default location of the zero frequency is in the upper left corner of the image rather than its center. This is not a mistake, it is how numpy handles the ordering of frequencies. Shifting has to be called explicitly if wanted.}

\begin{figure}[htpb]

import numpy as np		
img = read_image(...)
u = np.fft.fftfreq(img.shape[1])
v = np.fft.fftfreq(img.shape[0])
vv, uu = np.meshgrid(v, u, indexing='ij')
D_0 = 0.05
filter_f = (np.hypot(vv, uu) < D_0)
img_f = np.fft.fft2(img)
img_filtered = np.fft.ifft2(img_ft*filter_f).real
	\end{lstlisting}
	\end{tabular}
	\caption[Example listing]{Implementation and application of \gls{ilpf} (Equation \ref{eq:ilpf}). First, a coordinate system is generated with the discrete Fourier transform sample frequencies $u$ and $v$ matching the shape of the input image. Then, the function $np.meshgrid$ creates coordinate arrays for the $v$ and the $u$ coordinates and saves them into variable $vv$ and $uu$. Finally, \gls{ilpf} is calculated in frequency domain and stored in variable \emph{filter\textunderscore f}.}
\label{code:ilpf}
\end{figure}
